define monitor_imu
  while 1
    monitor halt
    printf "Roll: "
    call MahonyFilter.getRoll()
    printf " | Pitch: "
    call MahonyFilter.getPitch()
    printf " | Yaw: "
    call MahonyFilter.getYaw()

    monitor resume
    shell sleep 0.2
  end
end
